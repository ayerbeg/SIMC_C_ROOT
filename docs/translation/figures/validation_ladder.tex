  \node[rung=38mm, fill=simcblue!55, text=white] (l2) at (3.7,1.75)
    {\textbf{\small captures and replays}\\the whole chain on the Fortran's own random numbers

  \node[rung=38mm, fill=simcblue, text=white] (l3) at (7.4,3.5)
    {\textbf{\small RANLUX full runs}\\same seed, \texttt{-{}-fortran-compat}: \texttt{.hist}, \texttt{.geni},

  \node[simc/note, anchor=north west, text width=36mm] at (7.4,-0.15) {complete run, bit for bit};
  \node[simc/note, anchor=north west, text width=36mm] at (11.1,-0.15) {complete run, statistically};
  \draw[simc/arrow, draw=simcgray] (0,-0.75) -- (15.25,-0.75) node[pos=1, anchor=north east, simc/note] {more of the program in each test};
  % identity
  \node[simc/ok, anchor=north west, text width=62mm, align=left, font=\sffamily\scriptsize] at (0,5.4)
